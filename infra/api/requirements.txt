# SpeakLab API image.
#
# NO torch and NO speech model weights in this image. Every speech model lives in asr,
# tts or pron. api/tests/test_health.py asserts it rather than trusting this comment —
# a transitive dependency that quietly pulls torch would add ~2 GB to the image that
# has to start fastest. The one model that does live here is spaCy's small English
# pipeline, ~12 MB of statistical parser with no deep-learning runtime behind it.

fastapi==0.141.1
uvicorn[standard]==0.52.4
python-multipart==0.0.32

# ── Database ────────────────────────────────────────────────────────────────
# asyncpg is the request-path driver. psycopg2-binary is here for Alembic, which
# runs migrations synchronously — `postgresql+asyncpg://` is not a URL psycopg2 can
# parse, and config.py exposes SYNC_DATABASE_URL for exactly this.
sqlalchemy[asyncio]==2.0.52
asyncpg==0.31.0
psycopg2-binary==2.9.13
alembic==1.20.0

# ── Clients ─────────────────────────────────────────────────────────────────
# One async HTTP client for all four outbound callers: asr, tts, pron and Ollama. It is
# also the test client, so it was going to be here regardless.
httpx==0.28.1

# ── Auth ────────────────────────────────────────────────────────────────────
#
# Argon2id, through argon2-cffi rather than passlib. The product stores passwords
# Argon2-hashed. passlib is unmaintained, its bcrypt backend raises on bcrypt >= 4.1, and
# a wrapper whose value is switching between schemes buys nothing with one scheme and no
# legacy hashes to migrate; argon2-cffi gives hash/verify/check_needs_rehash directly.
# Importing users with other hashes is when passlib's CryptContext would earn its place.
#
# It installs from a wheel on linux/amd64 and linux/arm64, so no compiler is added to
# the image. Hashes carry the `$argon2id$` prefix at the library's default parameters
# (t=3, m=64 MiB, p=4).
argon2-cffi==25.1.0
pyjwt==2.14.0

# Pydantic's EmailStr, which is a FastAPI extra rather than a direct import. Hand-rolling
# an email regex is a well-known way to reject valid addresses, and the alternative of
# not validating at all makes the unique constraint on `users.email` the only thing
# standing between the table and two rows for the same person typed differently.
email-validator==2.3.0

# ── Analysis ────────────────────────────────────────────────────────────────
#
# The dependency parse and morphology behind deterministic grammar usage. spaCy brings
# numpy, thinc and blis and **no torch** — `tests/test_health.py` asserts that rather
# than trusting this comment, because a library that quietly acquired torch would add
# two gigabytes to the image that has to start fastest.
#
# The model is pinned by URL because it is not on PyPI. `python -m spacy download` would
# fetch whatever is current at build time, which makes two builds of the same commit
# different — and the parse is what every grammar number is computed from, so "the
# model changed" and "the learner changed" would be indistinguishable on a chart.
#
# The lemmatizer is not optional here. Without it `going`, `said` and `asked` never
# reach their base forms and every rule that names a verb silently stops firing.
spacy==3.8.16
https://github.com/explosion/spacy-models/releases/download/en_core_web_sm-3.8.0/en_core_web_sm-3.8.0-py3-none-any.whl

# ── Tests and lint ──────────────────────────────────────────────────────────
#
# In requirements-dev.txt, installed only into the image's `test` stage. The image the
# API runs from carries nothing it does not run.

# ── Deliberately absent ─────────────────────────────────────────────────────
#
# torch, transformers, faster-whisper, piper-tts  never. They belong to the model
#     services, and their absence is what keeps this image small enough to start in
#     seconds rather than multiple gigabytes.
#
# langchain, llamaindex  never. Orchestration here is a handful of HTTP calls and a
#     token budget, and a framework would hide the token budget — the part actually
#     worth reviewing.
