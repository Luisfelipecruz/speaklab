# SpeakLab ASR service.
#
# faster-whisper runs Whisper on CTranslate2, which is a C++ inference engine, not a
# training framework. That is the whole reason it is here: `openai-whisper` pulls torch
# and lands around 2.5 GB, and torch belongs in exactly one image in this system (pron,
# where forced alignment genuinely needs it). Note that faster-whisper's only
# reference to transformers/torch is inside its `conversion` extra, which is not
# installed here — this image converts nothing, it loads a pre-converted model.

faster-whisper==1.2.1

# PyAV decodes and resamples the upload. It is the ffmpeg libraries as a Python
# extension, so this image needs no ffmpeg *binary*, no subprocess and no temp file —
# and it is already a faster-whisper dependency, so pinning it costs nothing and gains
# the source metadata (container format, sample rate, channels) that the API stores on
# the audio_assets row.
av==18.1.0

# Imported directly, to hand CTranslate2 the float32 buffer it wants. It arrives
# transitively through ctranslate2 regardless; pinned because it is imported here.
numpy==2.5.3

fastapi==0.141.1
uvicorn[standard]==0.52.4
python-multipart==0.0.32
